cmake_minimum_required(VERSION 3.20)
project(iVivid VERSION 2.6.0 LANGUAGES NONE)
set(IVIVID_RELEASE_VERSION "${PROJECT_VERSION}-beta.1")
option(IVIVID_UPDATER_ONLY "Run updater regression tests without native Qt dependencies" OFF)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(IVIVID_DPKG_DEB dpkg-deb REQUIRED)
  find_program(IVIVID_OPENSSL openssl REQUIRED)
  add_test(NAME updater-debian-identity COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_deb_identity.py)
  add_test(NAME updater-selection COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_updater.py)
  add_test(NAME updater-authentication COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_release_auth.py)
  add_test(NAME updater-signed-roundtrip COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_signed_roundtrip.py)
  add_test(NAME updater-staging COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_release_stage.py)
  add_test(NAME updater-portable-stage COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_portable_stage.py)
  add_test(NAME updater-portable-activation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_portable_activate.py)
  add_test(NAME updater-preparation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/test_update_prepare.py)
  set_tests_properties(updater-debian-identity updater-selection updater-authentication updater-signed-roundtrip updater-staging updater-portable-stage updater-portable-activation updater-preparation PROPERTIES TIMEOUT 60)
endif()
if(IVIVID_UPDATER_ONLY)
  if(NOT BUILD_TESTING)
    message(FATAL_ERROR "IVIVID_UPDATER_ONLY requires BUILD_TESTING=ON")
  endif()
  message(STATUS "Updater-only test mode: no native build or release packaging")
  return()
endif()
enable_language(CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
add_executable(iVivid src/main.cpp src/MainWindow.cpp src/MainWindow.hpp src/Config.cpp src/Config.hpp)
target_link_libraries(iVivid PRIVATE Qt6::Widgets)
target_compile_definitions(iVivid PRIVATE IVIVID_RELEASE_VERSION="${IVIVID_RELEASE_VERSION}")
install(TARGETS iVivid RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS scripts/ivivid-play scripts/ivivid-update DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/iVivid.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
if(BUILD_TESTING)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
  add_executable(ivivid-gui-tests tests/gui_interaction_tests.cpp src/MainWindow.cpp src/MainWindow.hpp src/Config.cpp src/Config.hpp)
  target_link_libraries(ivivid-gui-tests PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME gui-interactions COMMAND ivivid-gui-tests)
  set_tests_properties(gui-interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;IVIVID_DISABLE_UPDATE_CHECK=1" TIMEOUT 30)
  add_executable(ivivid-config-tests tests/config_tests.cpp src/Config.cpp src/Config.hpp)
  target_link_libraries(ivivid-config-tests PRIVATE Qt6::Widgets)
  add_test(NAME config-tests COMMAND ivivid-config-tests)
  add_test(NAME cli-diagnostics COMMAND bash ${CMAKE_SOURCE_DIR}/tests/cli_tests.sh $<TARGET_FILE:iVivid> ${IVIVID_RELEASE_VERSION})
endif()
