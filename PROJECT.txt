# --- LONG DESCRIPTION (for README) ---
long_description: |
  iVivid turns your laptop's built-in graphics into a movie-night machine.
  It auto-detects your iGPU (Intel HD, UHD, Iris Xe, AMD Radeon), picks the
  perfect shader pipeline, and applies HDR-like contrast, vivid color,
  adaptive sharpening, and punchy audio - all in one click.
  No GPU tinkerers, no config files, no "which shader do I need?"
  Just open a video and watch it look dramatically better than it should
  on integrated graphics.

# --- PREREQUISITES ---
prerequisites:
  - Linux (XFCE preferred, GNOME/KDE work)
  - Python 3.8+
  - mpv 0.39+ (sudo apt install mpv)
  - SMPlayer (optional, sudo apt install smplayer)
  - Intel/AMD iGPU with VA-API support
  - python3-tk (sudo apt install python3-tk)
  - No pip packages required (stdlib only)

# --- INSTALLATION ---
installation:
  - "sudo apt install python3-tk mpv smplayer"
  - "git clone https://github.com/sudipdasin/ivivid.git"
  - "cd ivivid"
  - "chmod +x ivivid.py"
  - "python3 ivivid.py"

# --- USAGE ---
usage:
  step1: "Open GUI (double-click or python3 ivivid.py)"
  step2: "Select quality preset: LIGHT / BALANCED / MAX (one-click)"
  step3: "Tweak individual options via checkboxes (hover for help)"
  step4: "Click APPLY - writes mpv.conf + downloads shaders + disables compositor"
  step5: "Play video: mpv /path/to/video.mp4 or smplayer /path/to/video.mp4"

# --- FEATURES ---
features:
  video:
    - "480p to 1080p fullscreen, no bars, no crop"
    - "Pseudo-HDR look (S-curve contrast, gamma lift, saturation, warm tint, soft-clip)"
    - "Adaptive sharpening (edge-aware, no halos)"
    - "Chroma upscaling (KrigBilateral or CfL_Prediction)"
    - "Debanding (built-in + hdeband shader)"
    - "Dithering (temporal)"
    - "EWA Lanczos Sharp scaler + sigmoid upscaling"
    - "Hardware decode (VA-API)"
    - "Auto-play next in folder"
  audio:
    - "Bass boost (0-20dB slider)"
    - "Treble boost (0-20dB slider)"
    - "3D Surround (stereowiden: narrow/wide/ultra)"
    - "Compressor (dynamic range)"
    - "Loudness normalize (EBU R128)"
    - "Volume up to 150%"
  system:
    - "XFCE compositor auto-disable (biggest perf win)"
    - "iGPU auto-detection (HD 620, UHD 630, Iris Xe, AMD)"
    - "Quality presets auto-tuned per iGPU tier"

# --- QUALITY PRESETS ---
presets:
  light:
    description: "Minimum GPU load, 60fps guaranteed"
    shaders: "soft_look + adaptive-sharpen"
    scaling: "ewa_lanczossharp"
    hwdec: "vaapi"
    profile: "fast"
    best_for: "HD 620, UHD 630, older iGPUs"
  balanced:
    description: "Best quality/performance ratio"
    shaders: "vivid_hdr + CfL_Prediction + adaptive-sharpen"
    scaling: "ewa_lanczossharp + sigmoid"
    hwdec: "vaapi"
    profile: "fast"
    best_for: "HD 620, UHD 630, Iris Plus"
  max:
    description: "Maximum visual quality"
    shaders: "vivid_hdr + KrigBilateral + hdeband + adaptive-sharpen"
    scaling: "ewa_lanczossharp + sigmoid + correct-downscaling"
    hwdec: "vaapi"
    profile: "fast (off on Iris Xe)"
    best_for: "Iris Xe, AMD iGPU (700M+)"
